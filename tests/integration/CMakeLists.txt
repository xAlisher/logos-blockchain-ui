cmake_minimum_required(VERSION 3.21)
project(BlendNativeIntegration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Network RemoteObjects)
get_filename_component(DEFAULT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(BLEND_SOURCE "${DEFAULT_SOURCE}" CACHE PATH "Source matching the packaged plugin ABI")
set(BLEND_PLUGIN "" CACHE FILEPATH "Exact packaged native plugin .so")
set(LOGOS_QT_INCLUDE "" CACHE PATH "Directory containing logos_ui_plugin_context.h")
if(NOT EXISTS "${LOGOS_QT_INCLUDE}/logos_ui_plugin_context.h")
  message(FATAL_ERROR "Set LOGOS_QT_INCLUDE to the installed Logos Qt SDK cpp include directory")
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/rep_BlockchainBackend_source.h"
  COMMAND Qt6::repc -i rep -o source "${BLEND_SOURCE}/src/BlockchainBackend.rep" "${CMAKE_CURRENT_BINARY_DIR}/rep_BlockchainBackend_source.h"
  DEPENDS "${BLEND_SOURCE}/src/BlockchainBackend.rep")
# Deliberately NO backend .cpp, reducer copy, or backend moc here: all implementation
# and metaobject symbols must come from the actual packaged shared library.
add_library(harness_object OBJECT native_backend.cpp "${CMAKE_CURRENT_BINARY_DIR}/rep_BlockchainBackend_source.h")
target_compile_definitions(harness_object PRIVATE BLEND_RECOVERY_TICK=advanceBlendRecovery)
target_include_directories(harness_object PRIVATE "${BLEND_SOURCE}/src" "${CMAKE_CURRENT_BINARY_DIR}" "${LOGOS_QT_INCLUDE}")
target_link_libraries(harness_object PRIVATE Qt6::Core Qt6::Gui Qt6::Network Qt6::RemoteObjects)
if(BLEND_PLUGIN)
  if(NOT EXISTS "${BLEND_PLUGIN}")
    message(FATAL_ERROR "Packaged native plugin not found: ${BLEND_PLUGIN}")
  endif()
  add_library(packaged_backend SHARED IMPORTED)
  set_target_properties(packaged_backend PROPERTIES IMPORTED_LOCATION "${BLEND_PLUGIN}")
  add_executable(blend_native $<TARGET_OBJECTS:harness_object>)
  target_link_libraries(blend_native PRIVATE packaged_backend Qt6::Core Qt6::Gui Qt6::Network Qt6::RemoteObjects)
else()
  message(STATUS "Compile-only prerequisite check; no plugin supplied, integration execution is NOT available")
endif()
