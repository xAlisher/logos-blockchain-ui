cmake_minimum_required(VERSION 3.16)
project(BlendBackendTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core)
enable_testing()
add_executable(backend_blend_lifecycle ../backend_blend_lifecycle.cpp)
target_include_directories(backend_blend_lifecycle PRIVATE ../../src)
target_link_libraries(backend_blend_lifecycle PRIVATE Qt6::Core)
add_test(NAME backend_blend_lifecycle COMMAND backend_blend_lifecycle)
add_executable(backend_blend_transport ../backend_blend_transport.cpp)
target_include_directories(backend_blend_transport PRIVATE ../../src)
target_link_libraries(backend_blend_transport PRIVATE Qt6::Core)
add_test(NAME backend_blend_transport COMMAND backend_blend_transport)

foreach(name IN ITEMS backend_review backend_identity backend_mutation backend_http_timing)
    add_executable(${name} ../${name}.cpp)
    target_include_directories(${name} PRIVATE ../../src)
    target_link_libraries(${name} PRIVATE Qt6::Core)
endforeach()
add_test(NAME backend_review COMMAND backend_review)
add_test(NAME backend_identity COMMAND backend_identity)
add_test(NAME backend_mutation COMMAND backend_mutation)
add_test(NAME backend_http_timing COMMAND backend_http_timing ${CMAKE_CURRENT_SOURCE_DIR}/../fixtures/delayed_curl.py)
