cmake_minimum_required(VERSION 3.16)
project(BlendRecoveryReview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(recovery_policy native_policy.cpp)
target_include_directories(recovery_policy PRIVATE ../../src)
target_link_libraries(recovery_policy PRIVATE Qt6::Core)
enable_testing()
add_test(NAME recovery_policy COMMAND recovery_policy)
